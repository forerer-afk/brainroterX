const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "content-type",
};

function jsonResponse(data: any, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...corsHeaders,
      "Content-Type": "application/json",
    },
  });
}

function bytesToHex(bytes: Uint8Array) {
  return Array.from(bytes)
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}

async function hmacSha256(
  key: Uint8Array,
  data: string
): Promise<Uint8Array> {
  const cryptoKey = await crypto.subtle.importKey(
    "raw",
    key,
    {
      name: "HMAC",
      hash: "SHA-256",
    },
    false,
    ["sign"]
  );

  const signature = await crypto.subtle.sign(
    "HMAC",
    cryptoKey,
    new TextEncoder().encode(data)
  );

  return new Uint8Array(signature);
}

async function verifyTelegram(initData: string) {
  const botToken = Deno.env.get("BOT_TOKEN");

  if (!botToken) {
    throw new Error("BOT_TOKEN не найден");
  }

  const params = new URLSearchParams(initData);

  const receivedHash = params.get("hash");

  if (!receivedHash) {
    throw new Error("Telegram hash не найден");
  }

  params.delete("hash");

  const dataCheckString =
    [...params.entries()]
      .sort(([a], [b]) => a.localeCompare(b))
      .map(([key, value]) => `${key}=${value}`)
      .join("\n");

  const secretKey = await hmacSha256(
    new TextEncoder().encode("WebAppData"),
    botToken
  );

  const calculatedHash = bytesToHex(
    await hmacSha256(
      secretKey,
      dataCheckString
    )
  );

  if (calculatedHash !== receivedHash) {
    throw new Error("Неверные данные Telegram");
  }

  const userRaw = params.get("user");

  if (!userRaw) {
    throw new Error("Telegram user не найден");
  }

  return JSON.parse(userRaw);
}

function secureRandom01() {
  const data = new Uint32Array(1);

  crypto.getRandomValues(data);

  return data[0] / 4294967296;
}

function secureRandom100() {
  return secureRandom01() * 100;
}

function normalizeName(name: string) {
  return String(name || "")
    .trim()
    .toLowerCase();
}

async function isChannelMember(
  telegramId: number | string
) {
  const token = Deno.env.get("BOT_TOKEN");
  if (!token) {
    throw new Error("BOT_TOKEN не настроен");
  }
  const response = await fetch(
    `https://api.telegram.org/bot${token}/getChatMember?chat_id=@brainroterX_help&user_id=${telegramId}`
  );
  const data = await response.json();
  if (!response.ok || !data.ok) {
    console.log("getChatMember error", data);
    return false;
  }
  const status = String(data.result?.status || "");
  return ["creator","administrator","member"].includes(status);
}

async function sendAdminMessage(
  text: string,
  keyboard: any
) {
  const token = Deno.env.get("ADMIN_BOT_TOKEN");
  const chatId = Deno.env.get("ADMIN_CHAT_ID");

  if (!token || !chatId) {
    console.log(
      "ADMIN_BOT_TOKEN или ADMIN_CHAT_ID не настроены"
    );
    return;
  }

  try {
    await fetch(
      `https://api.telegram.org/bot${token}/sendMessage`,
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          chat_id: chatId,
          text,
          reply_markup: {
            inline_keyboard: keyboard,
          },
        }),
      }
    );
  } catch (error) {
    console.log(
      "Ошибка отправки сообщения админу",
      error
    );
  }
}

    console.log(

    await fetch(
      `https://api.telegram.org/bot${token}/sendMessage`,
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          chat_id: telegramId,
          text,
        }),
      }
    );
  } catch (error) {
    console.log(
      "Ошибка отправки сообщения игроку",
      error
    );
  }
}

function verifyAdminSecret(body: any) {
  const secret = Deno.env.get("ADMIN_SECRET");

  if (!secret) {
    throw new Error(
      "ADMIN_SECRET не настроен"
    );
  }

  if (body.admin_secret !== secret) {
    throw new Error("Нет доступа");
  }
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", {
      headers: corsHeaders,
    });
  }

  try {
    const body = await req.json();

    const action = String(
      body.action || "get_player"
    );

    const supabaseUrl =
      Deno.env.get("SUPABASE_URL");

    const serviceKey =
      Deno.env.get(
        "SUPABASE_SERVICE_ROLE_KEY"
      );

    if (!supabaseUrl) {
      throw new Error(
        "SUPABASE_URL не найден"
      );
    }

    if (!serviceKey) {
      throw new Error(
        "SUPABASE_SERVICE_ROLE_KEY не найден"
      );
    }

    const headers = {
      apikey: serviceKey,
      Authorization: `Bearer ${serviceKey}`,
      "Content-Type": "application/json",
    };


    /* =====================================================
       REFERRAL HELPERS
    ===================================================== */

    const generateReferralCode =
      () => {
        const alphabet =
          "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

        const bytes =
          new Uint8Array(6);

        crypto.getRandomValues(
          bytes
        );

        let result =
          "BR";

        for (
          const byte
          of bytes
        ) {
          result +=
            alphabet[
              byte %
              alphabet.length
            ];
        }

        return result;
      };


    const getOrCreateReferral =
      async (
        targetTelegramId: number
      ) => {
        const existingResponse =
          await fetch(
            `${supabaseUrl}/rest/v1/referral_codes?telegram_id=eq.${targetTelegramId}&select=*&limit=1`,
            {
              headers,
            }
          );

        const existing =
          await existingResponse.json();

        if (
          !existingResponse.ok
        ) {
          throw new Error(
            "Не удалось загрузить реферальный код"
          );
        }

        if (
          existing.length > 0
        ) {
          return existing[0];
        }

        for (
          let attempt = 0;
          attempt < 10;
          attempt++
        ) {
          const code =
            generateReferralCode();

          const createResponse =
            await fetch(
              `${supabaseUrl}/rest/v1/referral_codes`,
              {
                method:
                  "POST",

                headers: {
                  ...headers,

                  Prefer:
                    "return=representation",
                },

                body:
                  JSON.stringify({
                    telegram_id:
                      targetTelegramId,

                    code:
                      code,

                    referral_balance:
                      0,

                    total_earned:
                      0,

                    uses_count:
                      0,
                  }),
              }
            );

          const created =
            await createResponse.json();

          if (
            createResponse.ok &&
            created.length > 0
          ) {
            return created[0];
          }

          // Если случайный код уже занят —
          // просто пробуем другой.
          if (
            createResponse.status !==
            409
          ) {
            throw new Error(
              "Не удалось создать реферальный код"
            );
          }
        }

        throw new Error(
          "Не удалось сгенерировать уникальный реферальный код"
        );
      };


    /* =====================================================
       ADMIN — APPROVE BALANCE
    ===================================================== */

    if (
      action ===
      "admin_approve_balance_request"
    ) {
      verifyAdminSecret(body);

      const requestId =
        Number(body.request_id);

      if (!requestId) {
        throw new Error(
          "Не указан request_id"
        );
      }

      const requestResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?id=eq.${requestId}&status=eq.pending&select=*`,
          { headers }
        );

      const requests =
        await requestResponse.json();

      if (
        !requestResponse.ok ||
        requests.length === 0
      ) {
        throw new Error(
          "Заявка уже обработана или не найдена"
        );
      }

        requests[0];

      const approveResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?id=eq.${requestId}&status=eq.pending`,

            method:
              "PATCH",

              ...headers,

                status:

        !approveResponse.ok ||
        approved.length === 0
      ) {
        throw new Error(
          "Заявка уже была обработана"
        );
      }

      const playerResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/players?telegram_id=eq.${request.telegram_id}&select=id,coins,has_deposited,wager_required,wager_progress`,
          {
            headers,
          }
        );

      const players =
        await playerResponse.json();

      if (
        !playerResponse.ok ||
        players.length === 0
      ) {
        // Возвращаем заявку в pending,
        // чтобы админ мог повторить подтверждение.
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?id=eq.${requestId}`,
          {
            method:
              "PATCH",

            headers,

            body:
              JSON.stringify({
                status:
                  "pending",
              }),
          }
        );

        throw new Error(
          "Игрок не найден"
        );
      }

      const oldCoins =
        Number(
          players[0].coins ||
          0
        );

      const baseCoins =
        Math.max(
          0,
          Number(
            request.total_value ||
            0
          )
        );

      const referralBonus =
        request.referral_code
          ?
            Math.max(
              0,
              Math.floor(
                Number(
                  request.referral_bonus ||
                  0
                )
              )
            )
          :
            0;

      // Игрок получает обычное пополнение +10%,
      // если использовал действующий реферальный код.
      const addedCoins =
        baseCoins +
        referralBonus;

      const newCoins =
        oldCoins +
        addedCoins;


      // Отыгрыш x1.5 считается от всей суммы,
      // которая реально пришла игроку, включая +10%.
      const oldWagerRequired =
        Number(
          players[0].wager_required ||
          0
        );

      const oldWagerProgress =
        Number(
          players[0].wager_progress ||
          0
        );

      const addedWager =
        Math.max(
          0,
          addedCoins *
          1.5
        );

      const newWagerRequired =
        oldWagerRequired +
        addedWager;


      const coinsResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/players?telegram_id=eq.${request.telegram_id}`,
          {
            method:
              "PATCH",

            headers: {
              ...headers,

              Prefer:
                "return=representation",
            },

            body:
              JSON.stringify({
                coins:
                  newCoins,

                has_deposited:
                  true,

                wager_required:
                  newWagerRequired,

                wager_progress:
                  oldWagerProgress,
              }),
          }
        );

      if (
        !coinsResponse.ok
      ) {
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?id=eq.${requestId}`,

            method:
              "PATCH",

            headers,

            body:
              JSON.stringify({
                status:
                  "pending",
              }),
          }
        );

        throw new Error(
          "Не удалось начислить монеты"
        );
      }


      // Награда владельцу реферального кода
      // отправляется НЕ на основной баланс,
      // а в referral_balance. Он заберёт её
      // кнопкой "Получить монеты".
      //
      // Для SHARKBITE владелец получает 30%,
      // для остальных кодов — 10%.
      const normalizedReferralCode =
        String(
          request.referral_code ||
          ""
        )
        .trim()
        .toUpperCase();

      const referralOwnerBonus =
        normalizedReferralCode ===
        "SHARKBITE"
          ? Math.floor(
              baseCoins *
              0.30
            )
          : referralBonus;

      let referralRewardPaid =
        false;

      if (
        referralOwnerBonus > 0 &&
        request.referral_owner_telegram_id
      ) {
        const ownerTelegramId =
          Number(
            request.referral_owner_telegram_id
          );

        const ownerReferralResponse =
          await fetch(
            `${supabaseUrl}/rest/v1/referral_codes?telegram_id=eq.${ownerTelegramId}&select=*&limit=1`,
            {
              headers,
            }
          );

        const ownerRows =
          await ownerReferralResponse.json();

        if (
          ownerReferralResponse.ok &&
          ownerRows.length > 0
        ) {
          const ownerReferral =
            ownerRows[0];

          const newReferralBalance =
            Number(
              ownerReferral.referral_balance ||
              0
            ) +
            referralOwnerBonus;

          const newTotalEarned =
            Number(
              ownerReferral.total_earned ||
              0
            ) +
            referralOwnerBonus;

          const newUsesCount =
            Number(
              ownerReferral.uses_count ||
              0
            ) +
            1;

          const rewardResponse =
            await fetch(
              `${supabaseUrl}/rest/v1/referral_codes?telegram_id=eq.${ownerTelegramId}`,
              {
                method:
                  "PATCH",

                headers: {
                  ...headers,

                  Prefer:
                    "return=representation",
                },

                body:
                  JSON.stringify({
                    referral_balance:
                      newReferralBalance,

                    total_earned:
                      newTotalEarned,

                    uses_count:
                      newUsesCount,
                  }),
              }
            );

          referralRewardPaid =
            rewardResponse.ok;

          if (
            !rewardResponse.ok
          ) {
            console.log(
              "Не удалось начислить реферальную награду владельцу кода"
            );
          }
        }
      }


      return jsonResponse({
        ok:
          true,

        request_id:
          requestId,

        base_coins:
          baseCoins,

        referral_bonus:
          referralBonus,

        referral_owner_bonus:
          referralOwnerBonus,

        added_coins:
          addedCoins,

        coins:
          newCoins,

        has_deposited:
          true,

        wager_required:
          newWagerRequired,

        wager_progress:
          oldWagerProgress,

        wager_added:
          addedWager,

        referral_reward_paid:
          referralRewardPaid,
      });
    }


    /* =====================================================
       ADMIN — REJECT BALANCE
    ===================================================== */

    if (
      action ===
      "admin_reject_balance_request"
    ) {
      verifyAdminSecret(body);

      const requestId =
        Number(body.request_id);

      if (!requestId) {
        throw new Error(
          "Не указан request_id"
        );
      }

      const response =
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?id=eq.${requestId}&status=eq.pending`,
          {
            method: "PATCH",
            headers: {
              ...headers,
              Prefer: "return=representation",
            },
            body: JSON.stringify({
              status: "rejected",
            }),
          }
        );

      const changed =
        await response.json();

      if (
        !response.ok ||
        changed.length === 0
      ) {
        throw new Error(
          "Заявка уже обработана или не найдена"
        );
      }

      return jsonResponse({
        ok: true,
        request_id: requestId,
      });
    }


    /* =====================================================
       ADMIN — APPROVE WITHDRAW
    ===================================================== */

    if (
      action ===
      "admin_approve_withdraw_request"
    ) {
      verifyAdminSecret(body);

      const requestId =
        Number(body.request_id);

      if (!requestId) {
        throw new Error(
          "Не указан request_id"
        );
      }

      const requestResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}&status=eq.pending&select=*`,
          { headers }
        );

      const requests =
        await requestResponse.json();

      if (
        !requestResponse.ok ||
        requests.length === 0
      ) {
        throw new Error(
          "Заявка уже обработана или не найдена"
        );
      }

      const request =
        requests[0];

      const requestItems =
        Array.isArray(
          request.items
        )
        &&
        request.items.length > 0
        ?
        request.items
        :

            inventory_id:
              request.inventory_id,

              request.brainrot_name,

            (item: any) =>

                item.inventory_id

            (id: number) =>
              Number.isInteger(id)
              &&
              id > 0
          );

        inventoryIds.length < 1

          "В заявке нет Brainrot"
        );

        await fetch(
          `${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}&status=eq.pending`,
          {
            method: "PATCH",
            headers: {
              ...headers,
              Prefer: "return=representation",
            },
            body: JSON.stringify({
              status: "approved",
            }),
          }
        );

        await approveResponse.json();

      if (
        !approveResponse.ok ||
        changed.length === 0
      ) {
        throw new Error(
          "Заявка уже обработана"
        );
      }

      const deleteResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${request.telegram_id}&locked=eq.true&id=in.(${inventoryIds.join(",")})`,
          {
            method: "DELETE",
            headers: {
              ...headers,
              Prefer: "return=representation",
            },
          }
        );

        await deleteResponse.json();

      if (
        !deleteResponse.ok
        ||
        !Array.isArray(
          deleted

        ||
        deleted.length !==
        inventoryIds.length
      ) {

          `${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}`,
          {
            method: "PATCH",
            headers,
            body: JSON.stringify({
              status: "pending",
            }),

          "Не удалось удалить все Brainrot"
        );
      }

        requestItems
          .map(
            (item: any) =>
              item.brainrot_name
          )
          .join(
            ", "
          );

        request.telegram_id,

        ok: true,
        request_id: requestId,
        item_count:

       ADMIN — REJECT WITHDRAW

        [
          {

              request.inventory_id,

              request.brainrot_name,

      const inventoryIds =

        request_id: requestId,
        item_count:

    /* =====================================================
       ADMIN — OUT OF STOCK
    ===================================================== */

    if (
      action ===
      "admin_out_of_stock_withdraw_request"
    ) {
      verifyAdminSecret(body);

      const requestId =
        Number(body.request_id);

      if (!requestId) {
        throw new Error(
          "Не указан request_id"

      }

      const requestResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}&status=eq.pending&select=*`,

        );

      const requests =
        await requestResponse.json();

      if (
        !requestResponse.ok ||

            method:
              "PATCH",

            body:
              JSON.stringify({

                  claimAmount,

        );

        throw new Error(

        String(body.code || "")

          "Введите промокод"

        );

            method: "POST",

              telegram_id: telegramId,

        [];

        of selectedItems

        const brainrotId =
          Number(
            selected.brainrot_id

        const quantity =
          Math.floor(
            Number(
              selected.quantity ||
              0
            )

        if (
          !brainrotId ||
          quantity <= 0
        ) {
          continue;
        }

          brainrots.find(
            (item: any) =>
              Number(
                item.brainrot_id
              ) ===
              brainrotId
          );

        if (!brainrot) {
          throw new Error(
            "Один из Brainrot не найден"

        const baseValue =
          Number(
            brainrot.value ||
            0
          );

          !Number.isFinite(
            baseValue

          baseValue <= 0

          throw new Error(
            `Неверная цена ${brainrot.name}`

            1,
            baseValue - 8

            brainrot.name,

          quantity:
            quantity,

            valueEach,

        null;

Roblox: ${robloxUsername}
Telegram ID: ${telegramId}

${itemLines}

💰 База: ${totalValue} монет${

💰 Игрок получит: ${totalValue + referralBonus} монет`,

        ? new Date(player.daily_case_claimed_at).getTime()
        : 0;

      const remainingMs = Math.max(0, 24 * 60 * 60 * 1000 - elapsed);

      if (remainingMs > 0) {
        return jsonResponse({
          ok: false,
          error: "Ежедневный кейс уже получен",

        });

          "DELETE",

            "return=representation",

    );
  }

  const rewardItems:
    any[] = [];

  for (
    let i = 0;
    i < targetQuantity;
    i++

      telegram_id:
        telegramId,

      brainrot_id:
        targetItem.brainrot_id,

      brainrot_name:
        targetItem.name,

      rarity:
        targetItem.rarity,

        Number(
          targetItem.value

  }

      {
        method:
          "POST",

        headers: {
          ...headers,

          Prefer:
            "return=representation",
        },

        body:
          JSON.stringify(
            rewardItems
          ),
      }
    );

  const addedItems =
    await addResponse.json();

  if (
    !addResponse.ok ||
    addedItems.length !==
    targetQuantity
  ) {
    await fetch(
      `${supabaseUrl}/rest/v1/inventory`,
      {
        method:
          "POST",

        headers,

        body:
          JSON.stringify({
            telegram_id:
              telegramId,

            brainrot_id:
              sourceItem.brainrot_id,

            brainrot_name:
              sourceItem.brainrot_name,

            rarity:
              sourceItem.rarity,

            value:
              Number(
                sourceItem.value
              ),

            locked:
              false,
          }),
      }
    );

    throw new Error(
      "Не удалось выдать Brainrot после обмена"
    );
  }

  let newCoins =
    Number(
      player.coins
    );

  if (
    coinRemainder > 0
  ) {
    newCoins +=
      coinRemainder;

    const coinsResponse =
      await fetch(
        `${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,
        {
          method:
            "PATCH",

          headers: {
            ...headers,

            Prefer:
              "return=representation",
          },

          body:
            JSON.stringify({
              coins:
                newCoins,
            }),
        }
      );

    const changed =
      await coinsResponse.json();

    if (
      !coinsResponse.ok ||
      changed.length === 0
    ) {
      console.log(
        "Обмен выполнен, но не удалось начислить остаток монет"
      );
    }
  }

  return jsonResponse({
    ok:
      true,

    source_item:
      sourceItem,

    target_item:
      targetItem,

    target_quantity:
      targetQuantity,

    added_items:
      addedItems,

    added_coins:
      coinRemainder,

    coins:
      newCoins,
  });

if (
  action ===
  "mystery_upgrade"
) {
  const inventoryId =
    Number(
      body.inventory_id
    );

  if (!inventoryId) {
    throw new Error(
      "Не выбран Brainrot для Mystery"
    );
  }

  const sourceResponse =
    await fetch(
      `${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false&select=*`,
      {
        headers,
      }
    );

  const sourceItems =
    await sourceResponse.json();

  if (
    !sourceResponse.ok ||
    sourceItems.length === 0
  ) {
    throw new Error(
      "Brainrot не найден или ожидает вывода"
    );
  }

  const sourceItem =
    sourceItems[0];

  const sourceValue =
    Number(
      sourceItem.value
    );

  if (
    !Number.isFinite(
      sourceValue
    ) ||
    sourceValue <= 0
  ) {
    throw new Error(
      "Неверная стоимость исходного Brainrot"
    );
  }

  if (
    sourceValue < 30
  ) {
    throw new Error(
      "Минимальная ставка в Mystery — Brainrot стоимостью от 30 монет"
    );
  }

  const brainrotsResponse =
    await fetch(
      `${supabaseUrl}/rest/v1/brainrots?select=brainrot_id,name,value,rarity&order=value.asc`,
      {
        headers,
      }
    );

  const allBrainrots =
    await brainrotsResponse.json();

  if (
    !brainrotsResponse.ok ||
    !Array.isArray(
      allBrainrots
    ) ||
    allBrainrots.length === 0
  ) {
    throw new Error(
      "Не удалось загрузить список Brainrot"
    );
  }

  const roll =
    secureRandom100();

  let minMultiplier =
    0.35;

  let maxMultiplier =
    0.85;

  let tier =
    "bad";

  /*
    Mystery без x1, но хорошие множители редкие.

    72% -> x0.35–x0.85
    20% -> x1.10–x1.35
     6% -> x1.36–x1.65
     2% -> x1.66–x2.00

    Ровно x1 не выпадает.
  */

  if (
    roll < 72
  ) {
    minMultiplier =
      0.35;

    maxMultiplier =
      0.85;

    tier =
      "bad";
  }

  else if (
    roll < 92
  ) {
    minMultiplier =
      1.10;

    maxMultiplier =
      1.35;

    tier =
      "good";
  }

  else if (
    roll < 98
  ) {
    minMultiplier =
      1.36;

    maxMultiplier =
      1.65;

    tier =
      "great";
  }

  else {
    minMultiplier =
      1.66;

    maxMultiplier =
      2.00;

    tier =
      "jackpot";
  }

  const multiplierRoll =
    secureRandom01();

  const wantedMultiplier =
    minMultiplier +
    (
      maxMultiplier -
      minMultiplier
    ) *
    multiplierRoll;

  const wantedValue =
    sourceValue *
    wantedMultiplier;

  let targetItem:
    any = null;

  let bestDifference =
    Infinity;

  for (
    const brainrot of
    allBrainrots
  ) {
    const value =
      Number(
        brainrot.value
      );

    if (
      !Number.isFinite(
        value
      ) ||
      value <= 0
    ) {
      continue;
    }

    /*
      Никогда не выбираем Brainrot с той же стоимостью,
      что и исходный. Поэтому фактический результат
      Mystery тоже не сможет быть ровно x1.
    */
    if (
      value === sourceValue
    ) {
      continue;
    }

    const difference =
      Math.abs(
        value -
        wantedValue
      );

    if (
      difference <
      bestDifference
    ) {
      bestDifference =
        difference;

      targetItem =
        brainrot;
    }
  }

  if (!targetItem) {
    throw new Error(
      "Не удалось выбрать Mystery Brainrot"
    );
  }

  const deleteResponse =
    await fetch(
      `${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false`,
      {
        method:
          "DELETE",

        headers: {
          ...headers,

          Prefer:
            "return=representation",
        },
      }
    );

  const deletedItems =
    await deleteResponse.json();

  if (
    !deleteResponse.ok ||
    deletedItems.length === 0
  ) {
    throw new Error(
      "Не удалось использовать Brainrot в Mystery"
    );
  }

  const addResponse =
    await fetch(
      `${supabaseUrl}/rest/v1/inventory`,
      {
        method:
          "POST",

        headers: {
          ...headers,

          Prefer:
            "return=representation",
        },

        body:
          JSON.stringify({
            telegram_id:
              telegramId,

            brainrot_id:
              targetItem.brainrot_id,

            brainrot_name:
              targetItem.name,

            rarity:
              targetItem.rarity,

            value:
              Number(
                targetItem.value
              ),

            locked:
              false,
          }),
      }
    );

  const addedItems =
    await addResponse.json();

  if (
    !addResponse.ok ||
    addedItems.length === 0
  ) {
    await fetch(
      `${supabaseUrl}/rest/v1/inventory`,
      {
        method:
          "POST",

        headers,

        body:
          JSON.stringify({
            telegram_id:
              telegramId,

            brainrot_id:
              sourceItem.brainrot_id,

            brainrot_name:
              sourceItem.brainrot_name,

            rarity:
              sourceItem.rarity,

            value:
              Number(
                sourceItem.value
              ),

            locked:
              false,
          }),
      }
    );

    throw new Error(
      "Не удалось выдать Mystery Brainrot"
    );
  }

  const newUpgradesDone =
    Number(
      player.upgrades_done ||
      0
    ) + 1;

  await fetch(
    `${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,
    {
      method:
        "PATCH",

      headers,

      body:
        JSON.stringify({
          upgrades_done:
            newUpgradesDone,
        }),
    }
  );

  const actualMultiplier =
    Number(
      targetItem.value
    ) /
    sourceValue;

  return jsonResponse({
    ok:
      true,

    mystery:
      true,

    tier:
      tier,

    roll:
      roll,

    wanted_multiplier:
      Number(
        wantedMultiplier.toFixed(
          3
        )
      ),

    wanted_value:
      Number(
        wantedValue.toFixed(
          2
        )
      ),

    multiplier:
      Number(
        actualMultiplier.toFixed(
          2
        )
      ),

    source_item:
      sourceItem,

    item:
      addedItems[0],

    upgrades_done:
      newUpgradesDone,
  });
}

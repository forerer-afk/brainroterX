const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "content-type",
};

function jsonResponse(data: any, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...corsHeaders,
      "Content-Type": "application/json; charset=utf-8",
    },
  });
}

function bytesToHex(bytes: Uint8Array) {
  return Array.from(bytes)
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}

async function hmacSha256(
  key: Uint8Array,
  data: string
): Promise<Uint8Array> {
  const cryptoKey = await crypto.subtle.importKey(
    "raw",
    key,
    {
      name: "HMAC",
      hash: "SHA-256",
    },
    false,
    ["sign"]
  );

  const signature = await crypto.subtle.sign(
    "HMAC",
    cryptoKey,
    new TextEncoder().encode(data)
  );

  return new Uint8Array(signature);
}

async function verifyTelegram(initData: string) {
  const botToken = Deno.env.get("BOT_TOKEN");

  if (!botToken) {
    throw new Error("BOT_TOKEN не найден");
  }

  const params = new URLSearchParams(initData);

  const receivedHash = params.get("hash");

  if (!receivedHash) {
    throw new Error("Telegram hash не найден");
  }

  params.delete("hash");

  const dataCheckString =
    [...params.entries()]
      .sort(([a], [b]) => a.localeCompare(b))
      .map(([key, value]) => `${key}=${value}`)
      .join("\n");

  const secretKey = await hmacSha256(
    new TextEncoder().encode("WebAppData"),
    botToken
  );

  const calculatedHash = bytesToHex(
    await hmacSha256(
      secretKey,
      dataCheckString
    )
  );

  if (calculatedHash !== receivedHash) {
    throw new Error("Неверные данные Telegram");
  }

  const userRaw = params.get("user");

  if (!userRaw) {
    throw new Error("Telegram user не найден");
  }

  return JSON.parse(userRaw);
}

function secureRandom01() {
  const data = new Uint32Array(1);

  crypto.getRandomValues(data);

  return data[0] / 4294967296;
}

function secureRandom100() {
  return secureRandom01() * 100;
}

function normalizeName(name: string) {
  return String(name || "")
    .trim()
    .toLowerCase();
}

async function isChannelMember(
  telegramId: number | string
) {
  const userId =
    String(
      telegramId
    );

  /*
    Владелец проекта.
    Оставляем прямой пропуск для твоего Telegram ID.
  */
  if (
    userId ===
    "1273118871"
  ) {
    return true;
  }

  const channel =
    "@brainroterX_help";

  /*
    Сначала пробуем BOT_TOKEN основного бота.
    Если его нет/он не админ — пробуем ADMIN_BOT_TOKEN,
    если он также добавлен в Supabase Secrets.
  */
  const tokens =
    [
      Deno.env.get("BOT_TOKEN"),
      Deno.env.get("ADMIN_BOT_TOKEN"),
    ]
    .filter(
      (token): token is string =>
        Boolean(
          token &&
          token.trim()
        )
    );

  if (
    tokens.length === 0
  ) {
    throw new Error(
      "SUB_CHECK_NO_TOKEN"
    );
  }

  let lastError =
    "";

  for (
    const token
    of tokens
  ) {
    try {
      const response =
        await fetch(
          `https://api.telegram.org/bot${token}/getChatMember?chat_id=${encodeURIComponent(channel)}&user_id=${encodeURIComponent(userId)}`
        );

      const data =
        await response.json();

      if (
        !response.ok ||
        !data.ok
      ) {
        lastError =
          String(
            data?.description ||
            `HTTP ${response.status}`
          );

        console.log(
          "getChatMember failed:",
          lastError
        );

        continue;
      }

      const status =
        String(
          data.result?.status ||
          ""
        );

      if (
        status === "creator" ||
        status === "administrator" ||
        status === "member"
      ) {
        return true;
      }

      if (
        status === "restricted"
        &&
        data.result?.is_member === true
      ) {
        return true;
      }

      if (
        status === "left" ||
        status === "kicked"
      ) {
        return false;
      }

      return false;

    } catch (
      error
    ) {
      lastError =
        String(
          error instanceof Error
          ? error.message
          : error
        );

      console.log(
        "getChatMember exception:",
        lastError
      );
    }
  }

  throw new Error(
    `SUB_CHECK_FAILED:${lastError}`
  );
}

async function sendAdminMessage(
  text: string,
  keyboard: any
) {
  const token = Deno.env.get("ADMIN_BOT_TOKEN");
  const chatId = Deno.env.get("ADMIN_CHAT_ID");

  if (!token || !chatId) {
    console.log(
      "ADMIN_BOT_TOKEN или ADMIN_CHAT_ID не настроены"
    );
    return;
  }

  try {
    await fetch(
      `https://api.telegram.org/bot${token}/sendMessage`,
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json; charset=utf-8",
        },
        body: JSON.stringify({
          chat_id: chatId,
          text,
          reply_markup: {
            inline_keyboard: keyboard,
          },
        }),
      }
    );
  } catch (error) {
    console.log(
      "Ошибка отправки сообщения админу",
      error
    );
  }
}

function legacyDepositAllowed(brainrot: any, allBrainrots: any[]) {
  const name = normalizeName(brainrot?.name);
  const blocked = new Set([
    "la jolly grande","los tictacs","eviledon","gold gold gold",
    "swaggy bros","chillin chili","money money reindeer","guest 666",
    "los tacoritas","los spaghettis","w or l","stekini fatini","los primos"
  ]);
  const garama = (Array.isArray(allBrainrots) ? allBrainrots : []).find((item:any) => {
    const n = normalizeName(item?.name);
    return n.includes("garama") && n.includes("madundung");
  });
  const minimum = garama ? Number(garama.value || 0) : Number.POSITIVE_INFINITY;
  return Number(brainrot?.value || 0) >= minimum && !blocked.has(name);
}

function effectiveDepositEnabledIds(brainrots: any[], availabilityRows: any[]) {
  const map = new Map(
    (Array.isArray(availabilityRows) ? availabilityRows : []).map((row:any) => [Number(row.brainrot_id), row])
  );
  return (Array.isArray(brainrots) ? brainrots : [])
    .filter((item:any) => {
      const row:any = map.get(Number(item.brainrot_id));
      if (row && row.deposit_override === true) return true;
      if (row && row.deposit_override === false) return false;
      if (normalizeName(item?.rarity) === "gear") return true;
      return legacyDepositAllowed(item, brainrots);
    })
    .map((item:any) => Number(item.brainrot_id));
}

function effectiveWithdrawInStockIds(brainrots: any[], availabilityRows: any[]) {
  const map = new Map(
    (Array.isArray(availabilityRows) ? availabilityRows : []).map((row:any) => [Number(row.brainrot_id), row])
  );
  return (Array.isArray(brainrots) ? brainrots : [])
    .filter((item:any) => {
      const row:any = map.get(Number(item.brainrot_id));
      return !(row && row.withdraw_in_stock === false);
    })
    .map((item:any) => Number(item.brainrot_id));
}

async function getGlobalUpgradeMultiplier(
  supabaseUrl: string,
  headers: Record<string,string>
) {
  try {
    const response = await fetch(
      `${supabaseUrl}/rest/v1/game_admin_settings?key=eq.global_upgrade_multiplier&select=value&limit=1`,
      { headers }
    );
    if (!response.ok) return 1;
    const rows = await response.json();
    const raw = Array.isArray(rows) && rows.length ? Number(rows[0]?.value) : 1;
    return Number.isFinite(raw) ? Math.min(1.5, Math.max(0.5, raw)) : 1;
  } catch (_) {
    return 1;
  }
}

async function setGlobalUpgradeMultiplier(
  supabaseUrl: string,
  headers: Record<string,string>,
  multiplier: number
) {
  const value = Math.min(1.5, Math.max(0.5, Number(multiplier || 1)));
  const response = await fetch(
    `${supabaseUrl}/rest/v1/game_admin_settings?on_conflict=key`,
    {
      method:"POST",
      headers:{...headers,Prefer:"resolution=merge-duplicates,return=representation"},
      body:JSON.stringify({key:"global_upgrade_multiplier",value})
    }
  );
  if(!response.ok) throw new Error("Не удалось сохранить множитель шанса");
  return value;
}

async function fetchAdminTestAccount(
  supabaseUrl: string,
  headers: Record<string,string>,
  telegramId: number
) {
  try {
    const response = await fetch(
      `${supabaseUrl}/rest/v1/admin_test_accounts?telegram_id=eq.${telegramId}&select=telegram_id,enabled,sandbox_locked,upgrade_multiplier&limit=1`,
      { headers }
    );
    if (!response.ok) return { telegram_id:Math.floor(Number(telegramId) || 0), enabled:false, sandbox_locked:false, upgrade_multiplier:1 };
    const rows = await response.json();
    if (!Array.isArray(rows) || !rows.length) {
      return { telegram_id:Math.floor(Number(telegramId) || 0), enabled:false, sandbox_locked:false, upgrade_multiplier:1 };
    }
    const row = rows[0] || {};
    const requestedTelegramId = Math.floor(Number(telegramId));
    const rowTelegramId = Math.floor(Number(row?.telegram_id));

    // Strict safety guard: a TEST row can affect only the exact player it belongs to.
    // Even if PostgREST/cache ever returns an unexpected row, never reuse it for another player.
    if (
      !Number.isInteger(requestedTelegramId) || requestedTelegramId <= 0 ||
      !Number.isInteger(rowTelegramId) || rowTelegramId !== requestedTelegramId
    ) {
      return { telegram_id:requestedTelegramId, enabled:false, sandbox_locked:false, upgrade_multiplier:1 };
    }

    const multiplier = Number(row?.upgrade_multiplier ?? 1);
    return {
      telegram_id:rowTelegramId,
      enabled:Boolean(row?.enabled),
      sandbox_locked:Boolean(row?.sandbox_locked),
      upgrade_multiplier:Number.isFinite(multiplier)
        ? Math.min(60,Math.max(0,multiplier))
        : 1
    };
  } catch (_) {
    return { telegram_id:Math.floor(Number(telegramId) || 0), enabled:false, sandbox_locked:false, upgrade_multiplier:1 };
  }
}


async function getActivePersonalTestTargetId(
  supabaseUrl: string,
  headers: Record<string,string>
) {
  try {
    const response = await fetch(
      `${supabaseUrl}/rest/v1/game_admin_settings?key=eq.personal_test_target_id&select=value&limit=1`,
      { headers }
    );
    if (response.ok) {
      const rows = await response.json();
      if (Array.isArray(rows) && rows.length) {
        const target = Math.floor(Number(rows[0]?.value ?? 0));
        return Number.isInteger(target) && target > 0 ? target : 0;
      }
    }

    // Compatibility with old installs: if there is exactly one enabled TEST row,
    // treat only that row as the active personal target. Multiple enabled rows
    // intentionally resolve to 0 until an admin explicitly enables one target.
    const fallbackResponse = await fetch(
      `${supabaseUrl}/rest/v1/admin_test_accounts?enabled=eq.true&select=telegram_id&limit=2`,
      { headers }
    );
    if (!fallbackResponse.ok) return 0;
    const fallbackRows = await fallbackResponse.json();
    if (!Array.isArray(fallbackRows) || fallbackRows.length !== 1) return 0;
    const target = Math.floor(Number(fallbackRows[0]?.telegram_id ?? 0));
    return Number.isInteger(target) && target > 0 ? target : 0;
  } catch (_) {
    return 0;
  }
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", {
      headers: corsHeaders,
    });
  }

  try {
    const body = await req.json();

    const action = String(
      body.action || "get_player"
    );

    const supabaseUrl =
      Deno.env.get("SUPABASE_URL");

    const serviceKey =
      Deno.env.get(
        "SUPABASE_SERVICE_ROLE_KEY"
      );

    if (!supabaseUrl) {
      throw new Error(
        "SUPABASE_URL не найден"
      );
    }

    if (!serviceKey) {
      throw new Error(
        "SUPABASE_SERVICE_ROLE_KEY не найден"
      );
    }

    const headers = {
      apikey: serviceKey,
      Authorization: `Bearer ${serviceKey}`,
      "Content-Type": "application/json",
    };

    /* =====================================================
       BIG WINS PROFILE ENRICHMENT
       Keep the feed fast: only profile + counters are loaded here.
       Best drop and inventory are loaded only when a player opens
       the mini profile.
    ===================================================== */
    const enrichBigWinsRows = async (inputRows: any[]) => {
      let enriched = Array.isArray(inputRows) ? inputRows : [];

      const ids = [
        ...new Set(
          enriched
            .map((row: any) => Number(row.telegram_id || 0))
            .filter((id: number) => id > 0)
        ),
      ];

      if (!ids.length) {
        return enriched;
      }

      try {
        const [profilesResponse, playersResponse] = await Promise.all([
          fetch(
            `${supabaseUrl}/rest/v1/player_public_profiles?telegram_id=in.(${ids.join(",")})&select=telegram_id,first_name,last_name,username,photo_url`,
            { headers }
          ),
          fetch(
            `${supabaseUrl}/rest/v1/players?telegram_id=in.(${ids.join(",")})&select=telegram_id,cases_opened,upgrades_done`,
            { headers }
          ),
        ]);

        const profiles = profilesResponse.ok
          ? await profilesResponse.json()
          : [];
        const playersRows = playersResponse.ok
          ? await playersResponse.json()
          : [];

        const profileMap = new Map(
          (Array.isArray(profiles) ? profiles : []).map((p: any) => [
            Number(p.telegram_id),
            p,
          ])
        );
        const playerMap = new Map(
          (Array.isArray(playersRows) ? playersRows : []).map((p: any) => [
            Number(p.telegram_id),
            p,
          ])
        );

        enriched = enriched.map((row: any) => {
          const id = Number(row.telegram_id || 0);
          const profile: any = profileMap.get(id) || {};
          const stats: any = playerMap.get(id) || {};

          return {
            ...row,
            player_first_name: profile.first_name || "",
            player_last_name: profile.last_name || "",
            player_username: profile.username || "",
            player_photo_url: profile.photo_url || "",
            player_cases_opened: Number(stats.cases_opened || 0),
            player_upgrades_done: Number(stats.upgrades_done || 0),
          };
        });
      } catch (_) {}

      return enriched;
    };


    /* =====================================================
       REFERRAL HELPERS
    ===================================================== */

    const generateReferralCode =
      () => {
        const alphabet =
          "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

        const bytes =
          new Uint8Array(6);

        crypto.getRandomValues(
          bytes
        );

        let result =
          "BR";

        for (
          const byte
          of bytes
        ) {
          result +=
            alphabet[
              byte %
              alphabet.length
            ];
        }

        return result;
      };


    const getOrCreateReferral =
      async (
        targetTelegramId: number
      ) => {
        const existingResponse =
          await fetch(
            `${supabaseUrl}/rest/v1/referral_codes?telegram_id=eq.${targetTelegramId}&select=*&limit=1`,
            {
              headers,
            }
          );

        const existing =
          await existingResponse.json();

        if (
          !existingResponse.ok
        ) {
          throw new Error(
            "Не удалось загрузить реферальный код"
          );
        }

        const fixedReferralCode =
          targetTelegramId ===
          7491082648
            ? "WERIFCHIK"
            : (
                targetTelegramId ===
                6986400218
                  ? "PETROSYAN"
                  : (
                      targetTelegramId ===
                      1116312381
                        ? "LEMIKKSS"
                        : (
                            targetTelegramId ===
                            8335984895
                              ? "DANYATUFF"
                              : null
                          )
                    )
              );

        if (
          existing.length > 0
        ) {
          if (
            fixedReferralCode &&
            String(
              existing[0].code || ""
            )
            .trim()
            .toUpperCase() !==
            fixedReferralCode
          ) {
            const updateFixedResponse =
              await fetch(
                `${supabaseUrl}/rest/v1/referral_codes?telegram_id=eq.${targetTelegramId}`,
                {
                  method:
                    "PATCH",

                  headers: {
                    ...headers,

                    Prefer:
                      "return=representation",
                  },

                  body:
                    JSON.stringify({
                      code:
                        fixedReferralCode,
                    }),
                }
              );

            const updatedFixed =
              await updateFixedResponse.json();

            if (
              !updateFixedResponse.ok ||
              updatedFixed.length === 0
            ) {
              throw new Error(
                "Не удалось установить фиксированный реферальный код"
              );
            }

            return updatedFixed[0];
          }

          return existing[0];
        }

        for (
          let attempt = 0;
          attempt < 10;
          attempt++
        ) {
          const code =
            fixedReferralCode ||
            generateReferralCode();

          const createResponse =
            await fetch(
              `${supabaseUrl}/rest/v1/referral_codes`,
              {
                method:
                  "POST",

                headers: {
                  ...headers,

                  Prefer:
                    "return=representation",
                },

                    telegram_id:
                      targetTelegramId,

                    code:
                      code,

                    referral_balance:
                      0,

                    total_earned:
                      0,

                    uses_count:
                      0,
                  }),
              }
            );

          const created =
            await createResponse.json();

          if (
            createResponse.ok &&
            created.length > 0
          ) {
            return created[0];
          }

          // Если случайный код уже занят —
          // просто пробуем другой.
          if (
            createResponse.status !==
            409
          ) {
            throw new Error(
              "Не удалось создать реферальный код"
            );
          }
        }

        throw new Error(
          "Не удалось сгенерировать уникальный реферальный код"
        );
      };

    async function getHourlyDepositPromo() {
      const hourMs =
        60 * 60 * 1000;

      const now = Date.now();

      const hourIndex =
        Math.floor(now / hourMs);

      /*
        Stable for one clock hour, but impossible to predict from the client.
        SUPABASE_SERVICE_ROLE_KEY is already a server-only secret.
      */
      const seedBytes =
        new TextEncoder().encode(
          `${serviceKey}:brainroterx-hourly-deposit-promo:${hourIndex}`
        );

      const digest =
        new Uint8Array(
          await crypto.subtle.digest(
            "SHA-256",
            seedBytes
          )
        );

      /*
        Higher bonuses are intentionally rarer.
        5% is most common, 15% is the rarest.
      */
      const weightedPercents = [
        { percent: 5, weight: 20 },
        { percent: 6, weight: 18 },
        { percent: 7, weight: 16 },
        { percent: 8, weight: 14 },
        { percent: 9, weight: 12 },
        { percent: 10, weight: 10 },
        { percent: 11, weight: 8 },
        { percent: 12, weight: 6 },
        { percent: 13, weight: 4 },
        { percent: 14, weight: 2 },
        { percent: 15, weight: 1 },
      ];

      const totalWeight =
        weightedPercents.reduce(
          (sum, item) => sum + item.weight,
          0
        );

      const rollSeed =
        (
          digest[0] * 16777216 +
          digest[1] * 65536 +
          digest[2] * 256 +
          digest[3]
        ) >>> 0;

      let weightedRoll =
        rollSeed % totalWeight;

      let percent = 5;

      for (const item of weightedPercents) {
        if (weightedRoll < item.weight) {
          percent = item.percent;
          break;
        }

        weightedRoll -= item.weight;
      }

      const alphabet =
        "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

      let suffix = "";

      for (let index = 4; index < 11; index++) {
        suffix +=
          alphabet[
            digest[index] % alphabet.length
          ];
      }

      const expiresAtMs =
        (hourIndex + 1) * hourMs;

      return {
        code: `BRXH-${suffix}`,
        percent,
        expires_at:
          new Date(expiresAtMs).toISOString(),
        seconds_remaining:
          Math.max(
            0,
            Math.ceil(
              (expiresAtMs - now) / 1000
            )
          ),
      };
    }

    async function setAppSetting(
      key: string,
      value: boolean
    ) {
      const patchResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/app_settings?key=eq.${encodeURIComponent(key)}`,
          {
            method: "PATCH",
            headers: {
              ...headers,
              Prefer: "return=representation",
            },
            body: JSON.stringify({
              value,
            }),
          }
        );

      const changed =
        await patchResponse.json();

      if (!patchResponse.ok) {
        throw new Error(
          "Не удалось сохранить настройку"
        );
      }

      if (
        !Array.isArray(changed) ||
        changed.length === 0
      ) {
        const createResponse =
          await fetch(
            `${supabaseUrl}/rest/v1/app_settings`,
            {
              method: "POST",
              headers: {
                ...headers,
                Prefer: "return=representation",
              },
              body: JSON.stringify({
                key,
                value,
              }),
            }
          );

        if (!createResponse.ok) {
          throw new Error(
            "Не удалось создать настройку"
          );
        }
      }
    }

          `${supabaseUrl}/rest/v1/app_settings?key=eq.${encodeURIComponent(key)}&select=value&limit=1`,

        test_enabled:enabled,

    async function resolveDepositReferral(
      referralCodeRaw: string,
      totalValue: number,
      playerTelegramId: number
    ) {
      const referralCode =
        String(
          referralCodeRaw ||
          ""
        )
        .trim()
        .toUpperCase();

      let referralOwnerTelegramId:
        number | null =
        null;

      let referralBonus = 0;
      let referralOwnerBonus = 0;

      if (!referralCode) {
        return {
          referralCode: "",
          referralOwnerTelegramId,
          referralBonus,
          referralOwnerBonus,
          playerPercent: 0,
          ownerPercent: 0,
          isHourlyPromo: false,
        };
      }

      /*
        The hourly case-menu promo is a deposit bonus only.
        Nobody receives an owner/referral reward from it.
      */
      const hourlyPromo =
        await getHourlyDepositPromo();

      if (
        referralCode ===
        String(hourlyPromo.code || "")
          .trim()
          .toUpperCase()
      ) {
        const playerPercent =
          Number(hourlyPromo.percent || 0) / 100;

        referralBonus =
          Math.floor(
            totalValue *
            playerPercent
          );

        return {
          referralCode,
          referralOwnerTelegramId: null,
          referralBonus,
          referralOwnerBonus: 0,
          playerPercent,
          ownerPercent: 0,
          isHourlyPromo: true,
        };
      }

      let referralOwner: any = null;

      if (
        referralCode ===
        "LEMIKKSS"
      ) {
        referralOwnerTelegramId =
          1116312381;

        referralOwner =
          await getOrCreateReferral(
            1116312381
          );
      }
      else {
        const referralResponse =
          await fetch(
            `${supabaseUrl}/rest/v1/referral_codes?code=ilike.${encodeURIComponent(referralCode)}&select=*&limit=1`,
            {
              headers,
            }
          );

        const referralRows =
          await referralResponse.json();

        if (
          !referralResponse.ok
        ) {
          throw new Error(
            "Не удалось проверить реферальный код"
          );
        }

        if (
          referralRows.length === 0
        ) {
          throw new Error(
            "Реферальный код не найден или уже истёк"
          );
        }

        referralOwner =
          referralRows[0];

        referralOwnerTelegramId =
          Number(
            referralOwner.telegram_id
          );
      }

      if (
        referralOwnerTelegramId ===
        playerTelegramId
      ) {
        throw new Error(
          "Нельзя использовать свой реферальный код"
        );
      }

      const rates =
        getReferralPercents(
          referralCode,
          referralOwnerTelegramId
        );

      referralBonus =
        Math.floor(
          totalValue *
          rates.playerPercent
        );

      referralOwnerBonus =
        Math.floor(
          totalValue *
          rates.ownerPercent
        );

      return {
        referralCode,
        referralOwnerTelegramId,
        referralBonus,
        referralOwnerBonus,
        playerPercent:
          rates.playerPercent,
        ownerPercent:
          rates.ownerPercent,
        isHourlyPromo: false,
      };
    }

            brainrot.name,

          quantity:
            quantity,

          value_each:
            valueEach,

        preparedItems.length === 0

        throw new Error(
          "Нет выбранных Brainrot"
        );

      if (
        totalValue <= 0
      ) {
        throw new Error(
          "Неверная сумма пополнения"
        );
      }

        await resolveDepositReferral(
          referralCode,
          totalValue,
          telegramId

      const referralOwnerTelegramId =
        referral.referralOwnerTelegramId;

      const referralBonus =
        referral.referralBonus;

      const referralOwnerBonus =

      /*
        Антиспам заявок на пополнение.

        Если у игрока уже есть НЕОБРАБОТАННАЯ заявка,
        созданная меньше 5 минут назад, новую создать нельзя.

        Если админ подтвердил или отклонил заявку —
        status уже не pending, поэтому новую можно отправить сразу.

        Если админ ничего не нажал — через 5 минут

      */
      const cooldownMs =
        5 * 60 * 1000;

      const pendingResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?telegram_id=eq.${telegramId}&status=eq.pending&select=id,created_at&order=created_at.desc&limit=1`,

            headers,
          }

      const pendingRequests =
        await pendingResponse.json();

      if (!pendingResponse.ok) {
        throw new Error(
          "Не удалось проверить предыдущие заявки"
        );

      if (
        Array.isArray(pendingRequests) &&
        pendingRequests.length > 0

        const lastPending =
          pendingRequests[0];

          new Date(
            lastPending.created_at
          ).getTime();

        if (
          Number.isFinite(createdAt)
        ) {
          const elapsed =
            Date.now() -
            createdAt;

          if (
            elapsed <
            cooldownMs
          ) {
            const remainingMs =
              cooldownMs -
              elapsed;

            const remainingSeconds =
              Math.max(
                1,
                Math.ceil(
                  remainingMs / 1000
                )
              );

            const minutes =
              Math.floor(
                remainingSeconds / 60
              );

            const seconds =
              remainingSeconds % 60;

            const timeText =
              minutes > 0
                ? `${minutes} мин ${seconds} сек`
                : `${seconds} сек`;

            throw new Error(
              `У тебя уже есть заявка на пополнение. Повторную можно отправить через ${timeText}, либо сразу после подтверждения/отклонения текущей заявки.`
            );
          } else {
            await fetch(
              `${supabaseUrl}/rest/v1/balance_requests?id=eq.${lastPending.id}&status=eq.pending`,
              {method:"PATCH",headers,body:JSON.stringify({status:"rejected"})}

          }
        }
      }

      const requestResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests`,
          {
            method:
              "POST",

            headers: {
              ...headers,

              Prefer:
                "return=representation",
            },

            body:
              JSON.stringify({
                telegram_id:
                  telegramId,

                roblox_username:
                  robloxUsername,

                total_value:
                  totalValue,

                referral_code:
                  referralCode ||
                  null,

                referral_owner_telegram_id:
                  referralOwnerTelegramId,

                referral_bonus:
                  referralBonus,

                status:
                  "pending",
              }),
          }
        );

      const createdRequests =
        await requestResponse.json();

      if (
        !requestResponse.ok ||
        createdRequests.length === 0
      ) {
        throw new Error(
          "Не удалось создать заявку"
        );
      }

      const request =
        createdRequests[0];

      const requestItems =
        preparedItems.map(
          (item: any) => ({
            request_id:
              request.id,

            brainrot_id:
              item.brainrot_id,

            brainrot_name:
              item.brainrot_name,

            quantity:
              item.quantity,

            value_each:
              item.value_each,
          })
        );

        await fetch(
          `${supabaseUrl}/rest/v1/balance_request_items`,
          {
            method:
              "POST",

            headers: {
              ...headers,

              Prefer:
                "return=representation",
            },

            body:
              JSON.stringify(
                requestItems
              ),
          }

      if (
        !itemsResponse.ok
      ) {
        await fetch(
          `${supabaseUrl}/rest/v1/balance_requests?id=eq.${request.id}`,
          {
            method:
              "DELETE",

            headers,
          }
        );

        throw new Error(
          "Не удалось сохранить предметы заявки"

      }

      const itemLines =
        preparedItems
          .map(
            (item: any) =>
              `${item.brainrot_name} ×${item.quantity} = ${
                item.value_each *
                item.quantity
              } X`

          " X"
          :

        [
          [
            { text: "brainroterXbot", callback_data: `deposit_bot_0_${request.id}` },
            { text: "brainroterXbot1", callback_data: `deposit_bot_1_${request.id}` },
          ],
          [
            { text: "brainroterXbot2", callback_data: `deposit_bot_2_${request.id}` },
            { text: "brainroterXbot3", callback_data: `deposit_bot_3_${request.id}` },
          ],
          [
            { text: "brainroterXbot4", callback_data: `deposit_bot_4_${request.id}` },
          ],
          [
            {
              text:
                "\u2705 \u041F\u043E\u0434\u0442\u0432\u0435\u0440\u0434\u0438\u0442\u044C",

              callback_data:
                `deposit_approve_${request.id}`,

          ],

          [
            {
              text:
                "\u274C \u041E\u0442\u043A\u043B\u043E\u043D\u0438\u0442\u044C",

              callback_data:
                `deposit_reject_${request.id}`,

      return jsonResponse({
        ok:
          true,

        request_id:
          request.id,

        status:
          "pending",

        total_value:
          totalValue,

        referral_code:
          referralCode ||
          null,

        referral_bonus:
          referralBonus,

        referral_owner_bonus:
          referralOwnerBonus,

        credited_value:
          totalValue +
          referralBonus,

        items:
          preparedItems,

        created_at:
          request.created_at,

        trade_bot_username:
          request.trade_bot_username || null,

        message:
          "У вас есть 5 минут, чтобы зайти в игру и принять трейд",
      });
    }


    /* =====================================================
       PLAYER — CANCEL OWN PENDING WITHDRAW
    ===================================================== */

    if (action === "cancel_own_withdraw_request") {
      const requestId = Number(body.request_id);
      if (!Number.isInteger(requestId) || requestId <= 0) {
        throw new Error("Неверный номер заявки");
      }

      const requestResponse = await fetch(
        `${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}&telegram_id=eq.${telegramId}&status=eq.pending&select=*`,
        { headers }

      const requests = requestResponse.ok ? await requestResponse.json() : [];
      if (!requestResponse.ok || !Array.isArray(requests) || !requests.length) {
        throw new Error("Заявка уже обработана или не найдена");
      }
      const request = requests[0];
      const requestItems = Array.isArray(request.items) && request.items.length
        ? request.items
        : [{ inventory_id: request.inventory_id, brainrot_name: request.brainrot_name }];
      const inventoryIds = requestItems.map((x:any)=>Number(x.inventory_id)).filter((id:number)=>Number.isInteger(id)&&id>0);

        `${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}&telegram_id=eq.${telegramId}&status=eq.pending`,

          method:"PATCH",
          headers:{...headers,Prefer:"return=representation"},

        }

      const changed = statusResponse.ok ? await statusResponse.json() : [];
      if (!statusResponse.ok || !Array.isArray(changed) || !changed.length) {
        throw new Error("Заявка уже обработана");
      }

      if (inventoryIds.length) {
        const unlockResponse = await fetch(
          `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${telegramId}&id=in.(${inventoryIds.join(",")})`,
          {
            method:"PATCH",
            headers:{...headers,Prefer:"return=representation"},
            body:JSON.stringify({locked:false}),
          }
        );
        const unlocked = unlockResponse.ok ? await unlockResponse.json() : [];
        if (!unlockResponse.ok || !Array.isArray(unlocked) || unlocked.length !== inventoryIds.length) {
          await fetch(`${supabaseUrl}/rest/v1/withdraw_requests?id=eq.${requestId}`,{
            method:"PATCH",headers,body:JSON.stringify({status:"pending"})
          });
          throw new Error("Не удалось вернуть Brainrot в инвентарь");
        }
      }

      await sendAdminMessage(
        `↩️ ИГРОК ОТМЕНИЛ ЗАЯВКУ НА ВЫВОД #${requestId}\n\nTelegram ID: ${telegramId}\n📦 Brainrot возвращены игроку.\nЗаявка больше не активна.`
      );

      return jsonResponse({ok:true,request_id:requestId,inventory_ids:inventoryIds});
    }

    /* =====================================================
       CREATE WITHDRAW REQUEST
    ===================================================== */

    if (
      action ===
      "create_withdraw_request"
    ) {
      const withdrawEnabled =
        await getAppSetting(
          "withdraw_enabled",
          true
        );

      if (!withdrawEnabled) {
        throw new Error(
          "Вывод временно недоступен"
        );
      }

      const rawInventoryIds =
        Array.isArray(
          body.inventory_ids
        )
        ?
        body.inventory_ids
        :
        (
          body.inventory_id
          ?
          [body.inventory_id]
          :
          []
        );

      const inventoryIds =
        [
          ...new Set(
            rawInventoryIds
              .map(
                (id: unknown) =>
                  Number(id)
              )
              .filter(
                (id: number) =>
                  Number.isInteger(id)
                  &&
                  id > 0
              )
          ),
        ];

      const robloxUsername =
        String(
          body.roblox_username ||
          ""
        ).trim();

      const requestedTimeRaw =
        String(
          body.requested_time ||
          ""
        ).trim();

      if (
        inventoryIds.length < 1
        ||
        inventoryIds.length > 5
      ) {
        throw new Error(
          "Выберите от 1 до 5 Brainrot"
        );
      }

      if (
        robloxUsername.length < 3
      ) {
        throw new Error(
          "Введите Roblox ник"
        );
      }

      if (
        robloxUsername.length > 50
      ) {
        throw new Error(
          "Слишком длинный Roblox ник"
        );
      }

      if (
        !requestedTimeRaw
      ) {
        throw new Error(
          "Выберите дату и время"
        );
      }

      const requestedDate =
        new Date(
          requestedTimeRaw
        );

      if (
        Number.isNaN(
          requestedDate.getTime()
        )
      ) {
        throw new Error(
          "Неверная дата или время"
        );
      }

      if (
        requestedDate.getTime()
        <
        Date.now()
        -
        5 * 60 * 1000
      ) {
        throw new Error(
          "Выберите будущее время"
        );
      }

      const requestedTime =
        requestedDate.toISOString();

      const hasRealDeposit =
        Boolean(
          player.has_deposited
        )
        ||
        Number(
          player.wager_required ||
          0
        )
        >
        0;

      if (
        !hasRealDeposit
      ) {
        throw new Error(
          "Для вывода нужно сначала хотя бы один раз пополнить баланс Brainrot"
        );
      }

      const wagerRequired =
        Number(
          player.wager_required ||
          0
        );

      const wagerProgress =
        Number(
          player.wager_progress ||
          0
        );

      if (
        wagerProgress <
        wagerRequired
      ) {
        throw new Error(
          `Сначала выполни отыгрыш: ${wagerProgress} из ${wagerRequired} монет`
        );
      }

      const idFilter =
        inventoryIds.join(
          ","
        );

      const itemResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${telegramId}&locked=eq.false&id=in.(${idFilter})&select=*`,
          {
            headers,
          }
        );

      const items =
        await itemResponse.json();

      if (
        !itemResponse.ok
        ||
        !Array.isArray(
          items
        )
        ||
        items.length !==
        inventoryIds.length
      ) {
        throw new Error(
          "Один из Brainrot не найден или уже ожидает вывода"
        );
      }

      const withdrawAvailabilityRows =
        await fetchBrainrotAvailability(
          supabaseUrl,
          headers
        );

      const unavailableWithdrawItems =
        items.filter((item:any) => {
          const row = withdrawAvailabilityRows.find(
            (current:any) => Number(current.brainrot_id) === Number(item.brainrot_id)
          );
          return row && row.withdraw_in_stock === false;
        });

      if (unavailableWithdrawItems.length > 0) {
        return jsonResponse({
          ok:false,
          error_code:"WITHDRAW_OUT_OF_STOCK",
          error:`${unavailableWithdrawItems[0].brainrot_name} сейчас нет в наличии. Выбери замену.`,
          unavailable_items:unavailableWithdrawItems.map((item:any)=>({brainrot_id:item.brainrot_id,brainrot_name:item.brainrot_name,value:Number(item.value||0)}))
        });
      }

      const isBurguruWithdrawName = (name: unknown) => {
        const normalized = normalizeName(String(name || ""));
        return (
          normalized.includes("burguru")
          || normalized.includes("burguro")
        );
      };

      if (
        items.some(
          (item: any) =>
            Number(item.value || 0) < 59
            && !isBurguruWithdrawName(item.brainrot_name)
        )
      ) {
        throw new Error(
          "На вывод можно добавить Brainrot от 59 монет или Burguru"
        );
      }

      const lockResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${telegramId}&locked=eq.false&id=in.(${idFilter})`,
          {
            method: "PATCH",
            headers: {
              ...headers,
              Prefer: "return=representation",
            },
            body: JSON.stringify({
              locked: true,
            }),
          }
        );

      const lockedItems =
        await lockResponse.json();

      if (
        !lockResponse.ok
        ||
        !Array.isArray(
          lockedItems
        )
        ||
        lockedItems.length !==
        inventoryIds.length
      ) {
        if (
          Array.isArray(
            lockedItems
          )
          &&
          lockedItems.length > 0
        ) {
          const lockedIds =
            lockedItems
              .map(
                (item: any) =>
                  Number(
                    item.id
                  )
              )
              .filter(
                (id: number) =>
                  Number.isInteger(id)
                  &&
                  id > 0
              );

          if (
            lockedIds.length > 0
          ) {
            await fetch(
              `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${telegramId}&id=in.(${lockedIds.join(",")})`,
              {
                method: "PATCH",
                headers,
                body: JSON.stringify({
                  locked: false,
                }),
              }
            );
          }
        }

        throw new Error(
          "Не удалось зарезервировать все Brainrot"
        );
      }

      const orderedItems =
        inventoryIds
          .map(
            id =>
              items.find(
                (item: any) =>
                  Number(
                    item.id
                  )
                  ===
                  Number(
                    id
                  )
              )
          )
          .filter(
            Boolean
          );

      const firstItem =
        orderedItems[0];

      const requestItems =
        orderedItems.map(
          (item: any) => ({
            inventory_id:
              Number(
                item.id
              ),

            brainrot_id:
              item.brainrot_id,

            brainrot_name:
              item.brainrot_name,

            value:
              Number(
                item.value
              ),

            rarity:
              item.rarity,
          })
        );

      const requestResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/withdraw_requests`,
          {
            method: "POST",
            headers: {
              ...headers,
              Prefer: "return=representation",
            },
            body: JSON.stringify({
              telegram_id:
                telegramId,

              roblox_username:
                robloxUsername,

              // Старые поля оставляем для совместимости.
              inventory_id:
                firstItem.id,

              brainrot_id:
                firstItem.brainrot_id,

              brainrot_name:
                firstItem.brainrot_name,

              value:
                Number(
                  firstItem.value
                ),

              // Новый список всех выбранных Brainrot.
              items:
                requestItems,

              requested_time:
                requestedTime,

              status:
                "pending",
            }),
          }
        );

      const requests =
        await requestResponse.json();

      if (
        !requestResponse.ok
        ||
        !Array.isArray(
          requests
        )
        ||
        requests.length === 0
      ) {
        await fetch(
          `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${telegramId}&id=in.(${idFilter})`,
          {
            method: "PATCH",
            headers,
            body: JSON.stringify({
              locked: false,
            }),
          }
        );

        throw new Error(
          "Не удалось создать заявку на вывод"
        );

        requests[0];

      const formattedTime =
        requestedDate.toLocaleString(
          "ru-RU",
          {
            timeZone:
              "Europe/Kyiv",

            day:
              "2-digit",

            month:
              "2-digit",

            year:
              "numeric",

            hour:
              "2-digit",

            minute:
              "2-digit",

            hour12:
              false,
          }
        );

      const totalValue =
        requestItems.reduce(
          (
            sum: number,
            item: any

            sum
            +
            Number(

              ||
              0
            ),
          0
        );

        new Map<

              item.brainrot_name

                value:

            name

            1;

        Array.from(
          grouped.entries()
        )
        .map(
          (
            [
              name,
              info,

          ) =>
            `🧠 ${name} ×${info.quantity} — ${info.value * info.quantity}`
        )
        .join(
          "\n"
        );

`🔵 НОВАЯ ЗАЯВКА НА ВЫВОД #${request.id}

Roblox: ${robloxUsername}
Telegram ID: ${telegramId}

${itemLines}

📦 Количество: ${requestItems.length}
💰 Общая цена: ${totalValue}

🕒 Игрок будет в игре:
${formattedTime}`,
        [
          [
            {
              text:
                "✅ Подтвердить вывод",

              callback_data:
                `withdraw_approve_${request.id}`,
            },
          ],

          [
            {
              text:
                "❌ Отклонить",

              callback_data:
                `withdraw_reject_${request.id}`,
            },
          ],

          [
            {
              text:
                "📦 Нет в наличии",

              callback_data:
                `withdraw_out_of_stock_${request.id}`,

          ],

          [
            {
              text:
                "📴 Игрок не в сети",

              callback_data:
                `withdraw_player_offline_${request.id}`,

          ],

          [
            {
              text:
                "✏️ Неверный никнейм",

              callback_data:
                `withdraw_invalid_username_${request.id}`,

      return jsonResponse({
        ok: true,

        request_id:
          request.id,

        status:
          "pending",

        item_count:
          requestItems.length,

        items:
          requestItems,

        requested_time:
          requestedTime,
      });

       BIG WINS — GLOBAL LAST 15
    ===================================================== */

    if (
      action ===
      "get_big_wins"
    ) {
      const response = await fetch(
        `${supabaseUrl}/rest/v1/drop_history?value=gte.30&source=in.(case,daily,upgrade,upgrade_test)&select=id,telegram_id,brainrot_name,value,created_at,source,source_label&order=created_at.desc&limit=15`,
        { headers }
      );

      const rows = response.ok
        ? await response.json()
        : [];

      /*
        Previously this endpoint made an extra best-drop request for
        every player in the feed. That delayed avatars noticeably.
        The feed now needs only two parallel profile/stat queries.
      */
      const enriched =
        await enrichBigWinsRows(
          Array.isArray(rows) ? rows : []
        );

      return jsonResponse({
        ok: true,
        big_wins: enriched,
      });
    }


    /* =====================================================
       PUBLIC FEED PLAYER PROFILE
       Only players who have appeared in the public big-wins feed
       can be opened. Internal inventory IDs are never returned.
    ===================================================== */

    if (
      action ===
      "get_public_player_profile"
    ) {
      const targetTelegramId =
        Number(body.target_telegram_id || 0);

      if (
        !Number.isInteger(targetTelegramId)
        || targetTelegramId <= 0
      ) {
        throw new Error("Неверный игрок");
      }

      /* Keep this endpoint scoped to players who actually have feed drops. */
      const feedProofResponse = await fetch(
        `${supabaseUrl}/rest/v1/drop_history?telegram_id=eq.${targetTelegramId}&value=gte.30&source=in.(case,daily,upgrade,upgrade_test)&select=id&limit=1`,
        { headers }
      );
      const feedProofRows = feedProofResponse.ok
        ? await feedProofResponse.json()
        : [];

      if (
        !feedProofResponse.ok
        || !Array.isArray(feedProofRows)
        || feedProofRows.length === 0
      ) {
        throw new Error("Профиль игрока недоступен");
      }

      const [
        profileResponse,
        statsResponse,
        bestDropResponse,
        inventoryResponse,
      ] = await Promise.all([
        fetch(
          `${supabaseUrl}/rest/v1/player_public_profiles?telegram_id=eq.${targetTelegramId}&select=telegram_id,first_name,last_name,username,photo_url&limit=1`,
          { headers }
        ),
        fetch(
          `${supabaseUrl}/rest/v1/players?telegram_id=eq.${targetTelegramId}&select=telegram_id,cases_opened,upgrades_done&limit=1`,
          { headers }
        ),
        fetch(
          `${supabaseUrl}/rest/v1/drop_history?telegram_id=eq.${targetTelegramId}&brainrot_name=not.is.null&select=brainrot_name,value&order=value.desc,created_at.desc&limit=1`,
          { headers }
        ),
        fetch(
          `${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${targetTelegramId}&select=brainrot_id,brainrot_name,rarity,value,locked&order=value.desc&limit=200`,
          { headers }
        ),
      ]);

      const profileRows = profileResponse.ok
        ? await profileResponse.json()
        : [];
      const statsRows = statsResponse.ok
        ? await statsResponse.json()
        : [];
      const bestDropRows = bestDropResponse.ok
        ? await bestDropResponse.json()
        : [];
      const inventoryRows = inventoryResponse.ok
        ? await inventoryResponse.json()
        : [];

      const publicProfile =
        Array.isArray(profileRows) && profileRows.length
          ? profileRows[0]
          : {};
      const stats =
        Array.isArray(statsRows) && statsRows.length
          ? statsRows[0]
          : {};
      const bestDrop =
        Array.isArray(bestDropRows) && bestDropRows.length
          ? bestDropRows[0]
          : null;
      const publicInventory =
        Array.isArray(inventoryRows)
          ? inventoryRows
          : [];

      const inventoryTotalValue =
        publicInventory.reduce(
          (sum: number, item: any) =>
            sum + Math.max(0, Number(item.value || 0)),
          0
        );

      return jsonResponse({
        ok: true,
        player: {
          telegram_id: targetTelegramId,
          first_name: publicProfile.first_name || "",
          last_name: publicProfile.last_name || "",
          username: publicProfile.username || "",
          photo_url: publicProfile.photo_url || "",
          cases_opened: Number(stats.cases_opened || 0),
          upgrades_done: Number(stats.upgrades_done || 0),
          best_drop: bestDrop,
        },
        inventory: publicInventory.map((item: any) => ({
          brainrot_id: Number(item.brainrot_id || 0),
          brainrot_name: String(item.brainrot_name || ""),
          rarity: String(item.rarity || ""),
          value: Number(item.value || 0),
          locked: Boolean(item.locked),
        })),
        inventory_count: publicInventory.length,
        inventory_total_value: inventoryTotalValue,
        inventory_truncated: publicInventory.length >= 200,
      });
    }


    /* =====================================================
       DAILY CASE — ONCE PER 24H + CHANNEL SUBSCRIPTION
    ===================================================== */

    if (action === "daily_claim") {
      const lastClaim = player.daily_case_claimed_at
        ? new Date(player.daily_case_claimed_at).getTime()
        : 0;
      const elapsed = Math.max(0, Date.now() - lastClaim);
      const remainingMs = Math.max(0, 24 * 60 * 60 * 1000 - elapsed);

      if (remainingMs > 0) {
        return jsonResponse({
          ok: false,
          error: "Ежедневный кейс уже получен",
          remaining_seconds: Math.ceil(remainingMs / 1000),
        });
      }

      let subscribed = false;

      try {
        subscribed = await isChannelMember(telegramId);
      } catch (membershipError) {
        const membershipMessage =
          String(
            membershipError instanceof Error
            ? membershipError.message
            : membershipError
          );

        console.log(
          "daily membership check error",
          membershipMessage
        );

        return jsonResponse({
          ok: false,
          need_subscription: true,
          subscription_check_failed: true,
          error:
            membershipMessage.includes("SUB_CHECK_NO_TOKEN")
            ?
            "На сервере нет BOT_TOKEN для проверки подписки."
            :
            "Telegram не смог проверить подписку. Добавь основного бота администратором в @brainroterX_help.",
        });
      }

      if (!subscribed) {
        return jsonResponse({
          ok: false,
          need_subscription: true,
          error: "Подпишись на @brainroterX_help и нажми открыть ещё раз.",
        });
      }

      const nowIso = new Date().toISOString();
      const claimPatch = await fetch(
        `${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,
        {
          method: "PATCH",
          headers: { ...headers, Prefer: "return=representation" },
          body: JSON.stringify({ daily_case_claimed_at: nowIso }),
        }
      );
      const claimedRows = await claimPatch.json();
      if (!claimPatch.ok || claimedRows.length === 0) {
        throw new Error("Не удалось получить ежедневный кейс");
      }

      const roll = secureRandom100();
      let rewardType = "coins";
      let rewardAmount = 3;
      let rewardBrainrotName = "";

      if (roll < 79) { rewardAmount = 3; }
      else if (roll < 89) { rewardAmount = 5; }
      else if (roll < 94) { rewardAmount = 7; }
      else if (roll < 97) { rewardType = "brainrot"; rewardBrainrotName = "La Grande"; }
      else if (roll < 99) { rewardType = "brainrot"; rewardBrainrotName = "Nuclearo Dinossauro"; }
      else { rewardType = "brainrot"; rewardBrainrotName = "Money Money Puggy"; }

      if (rewardType === "coins") {
        const newCoins = Number(player.coins || 0) + rewardAmount;
        const coinPatch = await fetch(
          `${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,
          { method: "PATCH", headers: { ...headers, Prefer: "return=representation" }, body: JSON.stringify({ coins: newCoins }) }
        );
        if (!coinPatch.ok) {
          await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`, { method:"PATCH", headers, body:JSON.stringify({daily_case_claimed_at:null}) });
          throw new Error("Не удалось начислить монеты");
        }
        return jsonResponse({
          ok: true,
          reward_type: "coins",
          reward_amount: rewardAmount,
          coins: newCoins,
          daily: { available:false, remaining_seconds:86400, last_claimed_at:nowIso },
        });
      }

      const brainrotResponse = await fetch(
        `${supabaseUrl}/rest/v1/brainrots?name=ilike.${encodeURIComponent(rewardBrainrotName)}&select=brainrot_id,name,value,rarity&limit=1`,
        { headers }
      );
      const brainrotRows = await brainrotResponse.json();
      if (!brainrotResponse.ok || brainrotRows.length === 0) {
        await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`, { method:"PATCH", headers, body:JSON.stringify({daily_case_claimed_at:null}) });
        throw new Error(`Brainrot ${rewardBrainrotName} не найден`);
      }
      const reward = brainrotRows[0];
      const invResponse = await fetch(
        `${supabaseUrl}/rest/v1/inventory`,
        {
          method:"POST",
          headers:{...headers,Prefer:"return=representation"},
          body:JSON.stringify({telegram_id:telegramId,brainrot_id:reward.brainrot_id,brainrot_name:reward.name,rarity:reward.rarity,value:Number(reward.value),locked:false})
        }
      );
      const invRows = await invResponse.json();
      if (!invResponse.ok || invRows.length === 0) {
        await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`, { method:"PATCH", headers, body:JSON.stringify({daily_case_claimed_at:null}) });
        throw new Error("Не удалось добавить Brainrot");
      }
      await fetch(`${supabaseUrl}/rest/v1/drop_history`, {method:"POST",headers,body:JSON.stringify({telegram_id:telegramId,brainrot_id:reward.brainrot_id,brainrot_name:reward.name,value:Number(reward.value),source:"daily",source_label:"Daily Case"})});
      const hRes=await fetch(`${supabaseUrl}/rest/v1/drop_history?telegram_id=eq.${telegramId}&select=*&order=created_at.desc&limit=10`,{headers});
      const history=hRes.ok?await hRes.json():[];
      return jsonResponse({
        ok:true,reward_type:"brainrot",item:invRows[0],drop_history:history,
        daily:{available:false,remaining_seconds:86400,last_claimed_at:nowIso}
      });
    }


    /* =====================================================
       LADDER — SERVER-SIDE ANTI-ABUSE
    ===================================================== */

    if (action === "ladder_start") {
      const existingRes = await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?telegram_id=eq.${telegramId}&status=eq.active&select=id&limit=1`,{headers});
      const existing = existingRes.ok ? await existingRes.json() : [];
      if (existing.length > 0) throw new Error("Сначала закончи текущую лестницу");

      const stakeType = String(body.stake_type || "coins");
      let sessionPayload:any = {telegram_id:telegramId,stake_type:stakeType,current_step:0,status:"active"};

      if (stakeType === "coins") {
        const amount = Math.floor(Number(body.stake_amount || 0));
        if (amount < 30) throw new Error("Минимальная ставка — 30 X");
        const currentCoins = Number(player.coins || 0);
        if (currentCoins < amount) throw new Error("Недостаточно X Coin");
        const patch = await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}&coins=eq.${currentCoins}`,{method:"PATCH",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({coins:currentCoins-amount})});
        const changed=await patch.json(); if(!patch.ok||changed.length===0) throw new Error("Баланс изменился, попробуй ещё раз");
        sessionPayload={...sessionPayload,stake_value:amount};
      } else if (stakeType === "brainrot") {
        const inventoryId=Number(body.inventory_id||0); if(!inventoryId) throw new Error("Выбери Brainrot");
        const srcRes=await fetch(`${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false&select=*`,{headers});
        const srcRows=await srcRes.json(); if(!srcRes.ok||srcRows.length===0) throw new Error("Brainrot не найден");
        const src=srcRows[0]; if(Number(src.value)<30) throw new Error("Минимальная стоимость Brainrot — 30 X");
        const delRes=await fetch(`${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false`,{method:"DELETE",headers:{...headers,Prefer:"return=representation"}});
        const deleted=await delRes.json(); if(!delRes.ok||deleted.length===0) throw new Error("Не удалось поставить Brainrot");
        sessionPayload={...sessionPayload,stake_value:Number(src.value),inventory_id:inventoryId,brainrot_id:src.brainrot_id,brainrot_name:src.brainrot_name,brainrot_rarity:src.rarity};
      } else throw new Error("Неверный тип ставки");

      const createRes=await fetch(`${supabaseUrl}/rest/v1/ladder_sessions`,{method:"POST",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify(sessionPayload)});
      const rows=await createRes.json();
      if(!createRes.ok||rows.length===0){
        if(stakeType==="coins") await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,{method:"PATCH",headers,body:JSON.stringify({coins:Number(player.coins||0)})});
        else await fetch(`${supabaseUrl}/rest/v1/inventory`,{method:"POST",headers,body:JSON.stringify({telegram_id:telegramId,brainrot_id:sessionPayload.brainrot_id,brainrot_name:sessionPayload.brainrot_name,rarity:sessionPayload.brainrot_rarity,value:sessionPayload.stake_value,locked:false})});
        throw new Error("Не удалось создать лестницу");
      }
      const invRes=await fetch(`${supabaseUrl}/rest/v1/inventory?telegram_id=eq.${telegramId}&select=*&order=created_at.desc`,{headers});
      return jsonResponse({ok:true,coins:stakeType==="coins"?Number(player.coins||0)-sessionPayload.stake_value:Number(player.coins||0),inventory:invRes.ok?await invRes.json():[],ladder_session:rows[0]});
    }

    if (action === "ladder_step") {
      const sRes=await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?telegram_id=eq.${telegramId}&status=eq.active&select=*&order=id.desc&limit=1`,{headers});
      const rows=await sRes.json(); if(!sRes.ok||rows.length===0) throw new Error("Активная лестница не найдена");
      const s=rows[0]; const step=Number(s.current_step||0); if(step>=10) throw new Error("Ты уже на максимальной ступени — забери награду");
      const chances=[46.2,43.8,41,38.2,35,33,31,29,27,25]; const mult=[1.2,1.4,2,3.5,6,8,10,12,14,15];
      const claimRes=await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?id=eq.${s.id}&status=eq.active&current_step=eq.${step}`,{method:"PATCH",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({status:"processing"})});
      const claimed=await claimRes.json(); if(!claimRes.ok||claimed.length===0) throw new Error("Лестница уже обрабатывается");
      const win=secureRandom100()<chances[step];
      if(!win){
        // Отыгрыш засчитывается ТОЛЬКО при проигрыше ставки монетами.
        // Проигрыш Brainrot в лестнице отыгрыш не увеличивает.
        let ladderWagerProgress =
          Number(player.wager_progress || 0);

        const ladderWagerRequired =
          Number(player.wager_required || 0);

        if (
          s.stake_type === "coins" &&
          ladderWagerRequired > ladderWagerProgress
        ) {
          const lostCoins =
            Math.max(
              0,
              Math.floor(
                Number(s.stake_value || 0)
              )
            );

          ladderWagerProgress =
            Math.min(
              ladderWagerRequired,
              ladderWagerProgress + lostCoins
            );

          const wagerPatch =
            await fetch(
              `${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,
              {
                method:"PATCH",
                headers:{
                  ...headers,
                  Prefer:"return=representation"
                },
                body:JSON.stringify({
                  wager_progress:
                    ladderWagerProgress
                })
              }
            );

          if (!wagerPatch.ok) {
            // Не закрываем лестницу как проигранную, если отыгрыш
            // не удалось безопасно записать.
            await fetch(
              `${supabaseUrl}/rest/v1/ladder_sessions?id=eq.${s.id}&status=eq.processing`,
              {
                method:"PATCH",
                headers,
                body:JSON.stringify({
                  status:"active"
                })
              }
            );

            throw new Error(
              "Не удалось обновить отыгрыш"
            );
          }
        }

        await fetch(
          `${supabaseUrl}/rest/v1/ladder_sessions?id=eq.${s.id}&status=eq.processing`,
          {
            method:"PATCH",
            headers,
            body:JSON.stringify({
              status:"lost",
              finished_at:
                new Date().toISOString()
            })
          }
        );

        return jsonResponse({
          ok:true,
          win:false,
          current_step:step,
          wager_required:
            ladderWagerRequired,
          wager_progress:
            ladderWagerProgress,
          wager_remaining:
            Math.max(
              0,
              ladderWagerRequired -
              ladderWagerProgress
            )
        });
      }
      const nextStep=step+1;
      const updRes=await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?id=eq.${s.id}&status=eq.processing`,{method:"PATCH",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({status:"active",current_step:nextStep})});
      const upd=await updRes.json(); if(!updRes.ok||upd.length===0) throw new Error("Не удалось сохранить ступень");
      return jsonResponse({ok:true,win:true,current_step:nextStep,multiplier:mult[nextStep-1],ladder_session:upd[0]});
    }

    if (action === "ladder_cashout") {
      const sRes=await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?telegram_id=eq.${telegramId}&status=eq.active&select=*&order=id.desc&limit=1`,{headers});
      const rows=await sRes.json(); if(!sRes.ok||rows.length===0) throw new Error("Активная лестница не найдена");
      const s=rows[0]; const step=Number(s.current_step||0); if(step<1) throw new Error("Сначала пройди хотя бы одну ступень");
      const mult=[1.2,1.4,2,3.5,6,8,10,12,14,15]; const multiplier=mult[step-1];
      const claimRes=await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?id=eq.${s.id}&status=eq.active&current_step=eq.${step}`,{method:"PATCH",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({status:"cashing_out"})});
      const claimed=await claimRes.json(); if(!claimRes.ok||claimed.length===0) throw new Error("Награда уже обрабатывается");
      let addedItems:any[]=[]; let newCoins=Number(player.coins||0); let message="";
      if(s.stake_type==="coins"){
        const payout=Math.floor(Number(s.stake_value)*multiplier); newCoins+=payout;
        const p=await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,{method:"PATCH",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({coins:newCoins})}); if(!p.ok) throw new Error("Не удалось начислить X");
        message=`coin_cashout`;
      } else {
        const base=Number(s.stake_value); const targetValue=Math.floor(base*multiplier);
        if(step<=4){
          const brRes=await fetch(`${supabaseUrl}/rest/v1/brainrots?value=eq.${targetValue}&select=brainrot_id,name,value,rarity&limit=1`,{headers}); const brRows=await brRes.json();
          if(brRes.ok&&brRows.length>0){ const br=brRows[0]; const add=await fetch(`${supabaseUrl}/rest/v1/inventory`,{method:"POST",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({telegram_id:telegramId,brainrot_id:br.brainrot_id,brainrot_name:br.name,rarity:br.rarity,value:Number(br.value),locked:false})}); const a=await add.json(); if(!add.ok||a.length===0) throw new Error("Не удалось выдать Brainrot"); addedItems=a; message=`🧠 Получен ${br.name} за ${br.value} X`; }
          else { const add=await fetch(`${supabaseUrl}/rest/v1/inventory`,{method:"POST",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify({telegram_id:telegramId,brainrot_id:s.brainrot_id,brainrot_name:s.brainrot_name,rarity:s.brainrot_rarity,value:base,locked:false})}); const a=await add.json(); if(!add.ok) throw new Error("Не удалось вернуть Brainrot"); addedItems=a; const diff=Math.max(0,targetValue-base); newCoins+=diff; await fetch(`${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,{method:"PATCH",headers,body:JSON.stringify({coins:newCoins})}); message=`🧠 Brainrot возвращён + ${diff} X`; }
        } else {
          const qty=Math.floor(multiplier); const payload=Array.from({length:qty},()=>({telegram_id:telegramId,brainrot_id:s.brainrot_id,brainrot_name:s.brainrot_name,rarity:s.brainrot_rarity,value:base,locked:false}));
          const add=await fetch(`${supabaseUrl}/rest/v1/inventory`,{method:"POST",headers:{...headers,Prefer:"return=representation"},body:JSON.stringify(payload)}); const a=await add.json(); if(!add.ok||a.length!==qty) throw new Error("Не удалось выдать дубликаты"); addedItems=a; message=`🧠 Получено ${qty}× ${s.brainrot_name}`;
        }
      }
      // Любой Brainrot, реально полученный из Лестницы,
      // тоже может стать "Лучшим дропом".
      if(
        s.stake_type === "brainrot" &&
        Array.isArray(addedItems) &&
        addedItems.length > 0
      ){
        const bestLadderItem =
          addedItems.reduce(
            (best:any, item:any) =>
              Number(item?.value || 0) > Number(best?.value || 0)
                ? item
                : best,
            addedItems[0]
          );

        if(bestLadderItem){
          await fetch(
            `${supabaseUrl}/rest/v1/drop_history`,
            {
              method:"POST",
              headers,
              body:JSON.stringify({
                telegram_id:telegramId,
                brainrot_id:bestLadderItem.brainrot_id,
                brainrot_name:bestLadderItem.brainrot_name,
                value:Number(bestLadderItem.value || 0),
                source:"ladder",
                source_label:"Ladder"
              })
            }
          );
        }
      }

      await fetch(`${supabaseUrl}/rest/v1/ladder_sessions?id=eq.${s.id}&status=eq.cashing_out`,{method:"PATCH",headers,body:JSON.stringify({status:"cashed_out",finished_at:new Date().toISOString(),cashout_multiplier:multiplier})});
      return jsonResponse({
        ok:true,
        coins:newCoins,
        added_items:addedItems,
        multiplier,
        message,
        reward_type:
          s.stake_type === "coins"
          ? "coins"
          : "brainrot",
        payout_coins:
          s.stake_type === "coins"
          ? Math.floor(Number(s.stake_value) * multiplier)
          : 0,
      });
    }


    /* =====================================================
       OPEN CASE — NORMAL + DAILY DB CASE
    ===================================================== */

    if (
      action ===
      "open_case"
    ) {
      const caseId =
        Number(
          body.case_id
        );

      if (!caseId) {
        throw new Error(
          "Кейс не выбран"
        );
      }

      const caseResponse =
        await fetch(
          `${supabaseUrl}/rest/v1/cases?id=eq.${caseId}&select=*`,
          {
            headers,
          }
        );

      const caseRows =
        await caseResponse.json();

      if (
        !caseResponse.ok ||
        caseRows.length === 0
      ) {
        throw new Error(
          "Кейс не найден"
        );
      }

      const selectedCase =
        caseRows[0];

      const isDaily =
        Boolean(
          selectedCase.is_daily
        );

      const requestedOpenCount = Number(body.count || 1);
      const openCount = [1, 3, 5].includes(requestedOpenCount)
        ? requestedOpenCount
        : 1;

      if (isDaily && openCount !== 1) {
        throw new Error("Ежедневный кейс можно открыть только один раз");
      }

      const casePrice =
        isDaily
        ?
        0
        :
        Number(
          selectedCase.price ||
          0
        );

      if (
        !Number.isFinite(
          casePrice
        ) ||
        casePrice < 0
      ) {
        throw new Error(
          "Неверная цена кейса"
        );
      }

      const currentCoins =
        Number(
          player.coins ||
          0
        );

      if (
        !isDaily
        &&
        currentCoins <
        casePrice * openCount
      ) {
        throw new Error(
          "Недостаточно монет"
        );
      }

      if (
        isDaily
      ) {
        const lastClaim =
          player.daily_case_claimed_at
          ?
          new Date(
            player.daily_case_claimed_at

                  itemValue,

/* =====================================================
   DICE — BRAINROT x2 / x3
   Server result: 20% win.
   1 matching cube = x2 win.
   4 matching cubes = x3 win.
   2 or 3 matching cubes = loss.
===================================================== */
if (action === "dice_roll") {
  const inventoryId = Number(body.inventory_id);
  const selectedColor = String(body.color || "").trim().toLowerCase();

  const diceColors = [
    "red",
    "yellow",
    "blue",
    "orange",
    "green",
    "purple",
  ];

  if (!inventoryId) {
    throw new Error("Не выбран Brainrot");
  }

  if (!diceColors.includes(selectedColor)) {
    throw new Error("Не выбран цвет");
  }

  const sourceResponse = await fetch(
    `${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false&select=*`,
    { headers }
  );

  const sourceItems = await sourceResponse.json();

  if (!sourceResponse.ok || sourceItems.length === 0) {
    throw new Error("Brainrot не найден или ожидает вывода");
  }

  const sourceItem = sourceItems[0];

  // secureRandom100() is already used by the Upgrader backend.
  /*
    Overall Dice win chance = 20%.

    WIN:
    - exactly 1 selected-color cube
    - all 4 selected-color cubes

    LOSS:
    - 0 selected-color cubes
    - 2 selected-color cubes
    - 3 selected-color cubes

    The losing outcomes include 0 matches so the selected
    color no longer appears almost every roll.
  */
  const otherColors = diceColors.filter(
    (color) => color !== selectedColor
  );

  const win = secureRandom100() < 20;

  let matchCount = 0;

  if (win) {
    // 4 matching cubes are intentionally very rare.
    matchCount =
      secureRandom100() < 99 ? 1 : 4;
  } else {
    // Natural-looking losses:
    // 0 matches: 60%
    // 2 matches: 32%
    // 3 matches: 8%
    const lossRoll = secureRandom100();

    if (lossRoll < 60) {
      matchCount = 0;
    } else if (lossRoll < 92) {
      matchCount = 2;
    } else {
      matchCount = 3;
    }
  }

  const dice: string[] = [];

  for (let i = 0; i < matchCount; i++) {
    dice.push(selectedColor);
  }

  while (dice.length < 4) {
    const index = Math.floor(
      (secureRandom100() / 100) *
      otherColors.length
    );

    dice.push(
      otherColors[
        Math.min(
          index,
          otherColors.length - 1
        )
      ]
    );
  }

  // Shuffle the final cubes server-side.
  for (let i = dice.length - 1; i > 0; i--) {
    const j = Math.floor(
      (secureRandom100() / 100) *
      (i + 1)
    );

    const temp = dice[i];
    dice[i] = dice[j];
    dice[j] = temp;
  }

  // Consume exactly one selected inventory item.
  const deleteResponse = await fetch(
    `${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false`,
    {
      method: "DELETE",
      headers: {
        ...headers,
        Prefer: "return=representation",
      },
    }
  );

  const deleted = await deleteResponse.json();

  if (!deleteResponse.ok || deleted.length === 0) {
    throw new Error("Не удалось использовать Brainrot");
  }

  if (!win) {
    return jsonResponse({
      ok: true,
      win: false,
      chance: 20,
      match_count: matchCount,
      color: selectedColor,
      dice,
      source_item: sourceItem,
    });
  }

  // 1 matching cube = x2, all 4 matching cubes = x3.
  // The staked item was consumed above, so create 2 or 3 identical copies.
  const rewardMultiplier = matchCount === 4 ? 3 : 2;

  const rewardRows = Array.from({ length: rewardMultiplier }, () => ({
    telegram_id: telegramId,
    brainrot_id: sourceItem.brainrot_id,
    brainrot_name: sourceItem.brainrot_name,
    rarity: sourceItem.rarity,
    value: Number(sourceItem.value),
    locked: false,
  }));

  const rewardResponse = await fetch(
    `${supabaseUrl}/rest/v1/inventory`,
    {
      method: "POST",
      headers: {
        ...headers,
        Prefer: "return=representation",
      },
      body: JSON.stringify(rewardRows),
    }
  );

  const rewards = await rewardResponse.json();

  if (
    !rewardResponse.ok ||
    !Array.isArray(rewards) ||
    rewards.length < rewardMultiplier
  ) {
    // Restore the original stake if reward creation failed.
    await fetch(
      `${supabaseUrl}/rest/v1/inventory`,
      {
        method: "POST",
        headers,
        body: JSON.stringify({
          telegram_id: telegramId,
          brainrot_id: sourceItem.brainrot_id,
          brainrot_name: sourceItem.brainrot_name,
          rarity: sourceItem.rarity,
          value: Number(sourceItem.value),
          locked: false,
        }),
      }
    );

    throw new Error("Не удалось выдать выигрыш");
  }

  await fetch(
    `${supabaseUrl}/rest/v1/drop_history`,
    {
      method: "POST",
      headers,
      body: JSON.stringify({
        telegram_id: telegramId,
        brainrot_id: sourceItem.brainrot_id,
        brainrot_name: sourceItem.brainrot_name,
        value: Number(sourceItem.value),
        source: "dice",
        source_label: `Dice x${rewardMultiplier}`,
      }),
    }
  );

  return jsonResponse({
    ok: true,
    win: true,
    chance: 20,
    match_count: matchCount,
    color: selectedColor,
    dice,
    source_item: sourceItem,
    multiplier: rewardMultiplier,
    items: rewards,
  });
}

    );

  const allBrainrots =
    await brainrotsResponse.json();

  if (
    !brainrotsResponse.ok ||
    !Array.isArray(
      allBrainrots
    ) ||
    allBrainrots.length === 0
  ) {
    throw new Error(
      "Не удалось загрузить список Brainrot"
    );
  }

  const roll =
    secureRandom100();

  let minMultiplier =
    0.35;

  let maxMultiplier =
    0.85;

  let tier =
    "bad";

  /*
    Mystery без x1, но хорошие множители редкие.

    65% -> x0.35–x0.85
    27% -> x1.10–x1.35
     6% -> x1.36–x1.65
     2% -> x1.66–x2.00

    Ровно x1 не выпадает.
  */

  if (
    roll < 65
  ) {
    minMultiplier =
      0.35;

    maxMultiplier =
      0.85;

    tier =
      "bad";
  }

  else if (
    roll < 92
  ) {
    minMultiplier =
      1.10;

    maxMultiplier =
      1.35;

    tier =
      "good";
  }

  else if (
    roll < 98
  ) {
    minMultiplier =
      1.36;

    maxMultiplier =
      1.65;

    tier =
      "great";
  }

  else {
    minMultiplier =
      1.66;

    maxMultiplier =
      2.00;

    tier =
      "jackpot";
  }

  const multiplierRoll =
    secureRandom01();

  const wantedMultiplier =
    minMultiplier +
    (
      maxMultiplier -
      minMultiplier
    ) *
    multiplierRoll;

  const wantedValue =
    sourceValue *
    wantedMultiplier;

  let targetItem:
    any = null;

  let bestDifference =
    Infinity;

  for (
    const brainrot of
    allBrainrots
  ) {
    const value =
      Number(
        brainrot.value
      );

    if (
      !Number.isFinite(
        value
      ) ||
      value <= 0
    ) {
      continue;
    }

    /*
      Никогда не выбираем Brainrot с той же стоимостью,
      что и исходный. Поэтому фактический результат
      Mystery тоже не сможет быть ровно x1.
    */
    if (
      value === sourceValue
    ) {
      continue;
    }

    const difference =
      Math.abs(
        value -
        wantedValue
      );

    if (
      difference <
      bestDifference
    ) {
      bestDifference =
        difference;

      targetItem =
        brainrot;
    }
  }

  if (!targetItem) {
    throw new Error(
      "Не удалось выбрать Mystery Brainrot"
    );
  }

  const deleteResponse =
    await fetch(
      `${supabaseUrl}/rest/v1/inventory?id=eq.${inventoryId}&telegram_id=eq.${telegramId}&locked=eq.false`,
      {
        method:
          "DELETE",

        headers: {
          ...headers,

          Prefer:
            "return=representation",
        },
      }
    );

  const deletedItems =
    await deleteResponse.json();

  if (
    !deleteResponse.ok ||
    deletedItems.length === 0
  ) {
    throw new Error(
      "Не удалось использовать Brainrot в Mystery"
    );
  }

  const addResponse =
    await fetch(
      `${supabaseUrl}/rest/v1/inventory`,
      {
        method:
          "POST",

        headers: {
          ...headers,

          Prefer:
            "return=representation",
        },

        body:
          JSON.stringify({
            telegram_id:
              telegramId,

            brainrot_id:
              targetItem.brainrot_id,

            brainrot_name:
              targetItem.name,

            rarity:
              targetItem.rarity,

            value:
              Number(
                targetItem.value
              ),

            locked:
              false,
          }),
      }
    );

  const addedItems =
    await addResponse.json();

  if (
    !addResponse.ok ||
    addedItems.length === 0
  ) {
    await fetch(
      `${supabaseUrl}/rest/v1/inventory`,
      {
        method:
          "POST",

        headers,

        body:
          JSON.stringify({
            telegram_id:
              telegramId,

            brainrot_id:
              sourceItem.brainrot_id,

            brainrot_name:
              sourceItem.brainrot_name,

            rarity:
              sourceItem.rarity,

            value:
              Number(
                sourceItem.value
              ),

            locked:
              false,
          }),
      }
    );

    throw new Error(
      "Не удалось выдать Mystery Brainrot"
    );
  }

  const newUpgradesDone =
    Number(
      player.upgrades_done ||
      0
    ) + 1;

  // Mystery тоже считается полученным Brainrot для "Лучшего дропа".
  await fetch(
    `${supabaseUrl}/rest/v1/drop_history`,
    {
      method:
        "POST",

      headers,

      body:
        JSON.stringify({
          telegram_id:
            telegramId,

          brainrot_id:
            targetItem.brainrot_id,

          brainrot_name:
            targetItem.name,

          value:
            Number(
              targetItem.value
            ),

          source:
            "mystery",

          source_label:
            "Mystery Upgrader",
        }),
    }
  );

  await fetch(
    `${supabaseUrl}/rest/v1/players?telegram_id=eq.${telegramId}`,
    {
      method:
        "PATCH",

      headers,

      body:
        JSON.stringify({
          upgrades_done:
            newUpgradesDone,
        }),
    }
  );

  const actualMultiplier =
    Number(
      targetItem.value
    ) /
    sourceValue;

  return jsonResponse({
    ok:
      true,

    mystery:
      true,

    tier:
      tier,

    roll:
      roll,

    wanted_multiplier:
      Number(
        wantedMultiplier.toFixed(
          3
        )
      ),

    wanted_value:
      Number(
        wantedValue.toFixed(
          2
        )
      ),

    multiplier:
      Number(
        actualMultiplier.toFixed(
          2
        )
      ),

    source_item:
      sourceItem,

    item:
      addedItems[0],

    upgrades_done:
      newUpgradesDone,
  });
}


/* =====================================================
   UNKNOWN ACTION
===================================================== */

throw new Error(
  "Неизвестное действие"
);

  } catch (error) {
    console.error(
      error
    );

    const message =
      error instanceof Error
        ? error.message
        : String(
            error
          );

    return jsonResponse(
      {
        ok:
          false,

        error:
          message,
      },
      400
    );
  }
});
